cmake_minimum_required(VERSION 3.10)
project(drce_loc)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
add_executable(drce_loc_demo main.cpp drce_loc.cpp)
target_include_directories(drce_loc_demo PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(drce_loc_demo PRIVATE ${OpenCV_LIBS})
target_compile_options(drce_loc_demo PRIVATE -O2 -Wall)
